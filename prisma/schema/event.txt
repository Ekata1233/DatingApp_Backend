

model Event {
  id                   String             @id @default(uuid()) @db.Uuid

  // Business
  businessDetailsId    String?            @db.Uuid
  businessDetails      BusinessDetails?   @relation(fields: [businessDetailsId], references: [id])

  // Step 1 - Basics
  eventDurationType    EventDurationType?
  tripType             TripType?
  categoryId           String?            @db.Uuid
  category             EventCategory?     @relation(fields: [categoryId], references: [id])
  title                String?
  tagline              String?
  about                String?
  visibility           EventVisibility?

  // Step 2 - Photos
  highlightsCaption    String?

  // Experience
  experienceSubtitle   String?

  // Tickets & Crowd
  price                Decimal?           @db.Decimal(12, 2)
  originalPrice        Decimal?           @db.Decimal(12, 2)
  priceType            EventPriceType      @default(PER_PERSON)
  totalSeats           Int?
  maxTicketsPerPerson  Int                @default(1)
  crowdMix             CrowdMix?
  minAge               Int?
  maxAge               Int?
  groupSize            Int?
  intent               EventIntent?
  waitlistEnabled      Boolean            @default(true)

  // Included & Packing
  included             String[]           @default([])
  notIncluded          String[]           @default([])
  whatToBring          String[]           @default([])
  dressCode            EventDressCode     @default(NONE)

  // Host / Trip Captain
  hostPhotoUrl         String?
  hostTitle            String?
  hostSpeciality       String?
  hostLanguages        String[]           @default([])
  hostMessage          String?
  afterEventLine       String?

  // FAQs & Policies
  cancellationPolicy   EventCancellationPolicy?
  rules                String[]           @default([])
  safetyOptions        EventSafetyOption[] @default([])

  // Draft & Submission
  status               EventStatus        @default(DRAFT)
  currentStep          Int                @default(1)
  completedSteps       Int[]              @default([])
  submittedAt          DateTime?
  approvedAt           DateTime?
  publishedAt          DateTime?
  rejectionReason      String?

  // Relations
  photos               EventPhoto[]
  dates                EventDate[]
  venue                EventVenue?
  experience           EventExperience[]
  tripDetails          TripDetails?
  faqs                 EventFAQ[]
  hosts                EventHost[]

  createdAt            DateTime           @default(now())
  updatedAt            DateTime           @updatedAt

  @@index([businessDetailsId])
  @@index([categoryId])
  @@index([status])
  @@index([visibility, status])
  @@index([eventDurationType, status])
  @@index([businessDetailsId, status, updatedAt])

  @@map("events")
}


model EventPhoto {
  id          String         @id @default(uuid()) @db.Uuid
  eventId     String         @db.Uuid
  event       Event          @relation(fields: [eventId], references: [id], onDelete: Cascade)
  photoUrl    String
  fileId      String?
  photoType   EventPhotoType
  sortOrder   Int

  createdAt   DateTime       @default(now())
  updatedAt   DateTime       @updatedAt

  @@unique([eventId, photoType, sortOrder])
  @@map("event_photos")
}

model EventDate {
  id               String           @id @default(uuid()) @db.Uuid
  eventId          String           @db.Uuid
  event            Event            @relation(fields: [eventId], references: [id], onDelete: Cascade)
  startDateTime    DateTime
  endDateTime      DateTime?
  gateOpensAt      DateTime?
  timezone         String           @default("Asia/Kolkata")
  totalSeats       Int?
  womenSeats       Int?
  menSeats         Int?
  status           EventDateStatus  @default(SCHEDULED)
  isPrimary        Boolean          @default(false)
  cancelledAt      DateTime?
  cancellationReason String?

  createdAt        DateTime         @default(now())
  updatedAt        DateTime         @updatedAt

  @@unique([eventId, startDateTime])
  @@index([eventId, status])
  @@index([status, startDateTime])
  @@map("event_dates")
}

model TripDetails {
  id                   String       @id @default(uuid()) @db.Uuid
  eventId              String       @unique @db.Uuid
  event                Event        @relation(fields: [eventId], references: [id], onDelete: Cascade)
  // Dates & Route
  totalDays            Int          @default(2)
  travelBy             TravelType?
  // Pickup
  pickupCity           String?
  pickupPoint          String?
  exactPickupSpot      String?
  pickupMapsLink       String?
  pickupLatitude       Float?
  pickupLongitude      Float?
  // Destination
  destinationPlace     String?
  destinationArea      String?
  journeyThere         String?
  journeyBack          String?
  destinationIcon      String?
  // Drop
  dropCity             String?
  dropPoint            String?
  // Rooms
  roomSharing          RoomSharing?
  stayTile             String?
  mealsTile            String?
  // Ride Ops - Host Only
  vehiclePhotos        String[]     @default([])
  vehicleOperator      String?
  vehicleModel         String?
  vehicleCapacity      Int?
  vehicleSpecs         String[]     @default([])
  onBoard              String[]     @default([])
  safetyNoteTitle      String?
  safetyNote           String?
  // Relations
  days                 TripDay[]
  stays                TripStay[]
  places               TripPlace[]
  roomOptions          TripRoomOption[]

  createdAt            DateTime     @default(now())
  updatedAt            DateTime     @updatedAt

  @@map("trip_details")
}

model TripDay {
  id              String       @id @default(uuid()) @db.Uuid

  dayNumber       Int
  headline        String?
  photoUrl        String?
  highlightPhotos String[]     @default([])
  mealsIncluded   MealType[]   @default([])
  level           TripLevel?
  // Night Stay
  nightType       TripNightType?
  stayId          String?      @db.Uuid
  stay            TripStay?    @relation(fields: [stayId], references: [id], onDelete: Restrict)
  // Relations
  stops           TripStop[]
  games           TripGame[]

  createdAt       DateTime     @default(now())
  updatedAt       DateTime     @updatedAt

  @@unique([tripDetailsId, dayNumber])
  @@index([stayId])
  @@map("trip_days")
}
